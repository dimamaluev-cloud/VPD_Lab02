=== 8-bit device register ===
start                  dec =   0, hex = 0x00, bin = 00000000
set POWER              dec =   1, hex = 0x01, bin = 00000001
set READY              dec =   3, hex = 0x03, bin = 00000011
set TX                 dec =  11, hex = 0x0B, bin = 00001011
clear READY            dec =   9, hex = 0x09, bin = 00001001
toggle ERROR (1st)     dec =  13, hex = 0x0D, bin = 00001101
toggle ERROR (2nd)     dec =   9, hex = 0x09, bin = 00001001

=== test bits ===
POWER set? 1
READY set? 0
ERROR set? 0
TX    set? 1
RX    set? 0

=== mask as 1u << n ===
n = 4, mask = 1u << n = 16 (hex 0x10), equals REG_RX: 1
set bit n=4 (RX)       dec =  25, hex = 0x19, bin = 00011001
